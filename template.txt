# ========================================================
# Template Script Registrasi ONU ke ZTE OLT
# File ini dibaca otomatis setiap kali tombol aksi 'Reg' ditekan.
# Anda dapat mengedit baris konfigurasi di bawah ini sesuai kebutuhan jaringan.
#
# Variabel yang tersedia dan otomatis digantikan oleh sistem:
# - [[card]] atau [[slot]]              : Nomor card/slot OLT (contoh: 3)
# - [[port]]                            : Nomor port GPON OLT (contoh: 1)
# - [[onu]] atau [[onu_id]]             : Nomor ONU ID otomatis dari OLT (contoh: 4)
# - [[sn]]                              : Serial Number modem (contoh: ZTEGCE94FC33)
# - [[type]] atau [[onu_type]]          : Tipe profil ONU (contoh: ZTEG-F609)
# - 1441 atau 1441              : VLAN ID pelanggan (contoh: 100)
# - BVISION-A1-[[onu]] atau [[username]]: Username PPPoE pelanggan
# - 12345678 atau [[password]]: Password PPPoE pelanggan
# - [[interface]]                       : Interface GPON (contoh: gpon-olt_1/3/1)
# - [[onu_interface]]                   : Interface ONU lengkap (contoh: gpon-onu_1/3/1:4)
#
# Catatan: Baris yang diawali dengan tanda '#' atau '//' adalah komentar
# dan TIDAK akan dikirimkan ke OLT.
# ========================================================

config t

interface gpon-olt_1/[[card]]/[[port]]
    onu [[onu]] type ZTEG-F609 sn [[sn]]
    exit
!
interface gpon-onu_1/[[card]]/[[port]]:[[onu]]
    tcont 4 profile TR069NEW-10MB
    gemport 4 tcont 4
    gemport 4 traffic-limit downstream TR069NEW-10MB
    service-port 4 vport 4 user-vlan 1333 vlan 1333
    name BVISION-A1-[[onu]]
    sn-bind enable sn
    tcont 1 name INTERNET profile 1GB
    gemport 1 name INTERNET tcont 1
    gemport 1 traffic-limit upstream DOWN-1GB downstream DOWN-1GB 
    service-port 1 vport 1 user-vlan 1441 vlan 1441 
    service-port 1 description INTERNET
    exit
!
pon-onu-mng gpon-onu_1/[[card]]/[[port]]:[[onu]]
    service INTERNET gemport 1 vlan 1441
    wan-ip 1 mode pppoe username BVISION-A1-[[onu]] password 12345678 vlan-profile INTERNET-C1441 host 1
    wan-ip 1 ping-response enable traceroute-response enable
    security-mgmt 212 state enable mode forward protocol web
    vlan port eth_0/1 mode tag vlan 1441
    dhcp-ip ethuni eth_0/1 from-onu
    wan 1 ethuni 1 ssid 1 service internet host 1
    firewall enable level low anti-hack disable
    service TR069NEW gemport 4 vlan 1333
    wan-ip 2 mode DHCP vlan-profile TR069NEW
    veip 1 port udp 1232 host 2
    tr069-mgmt 1 state unlock
    tr069-mgmt 1 acs http://10.99.16.16:7547 validate basic username bbn password bbnet
    tr069-mgmt 1 tag pri 2 vlan 1333
    firewall enable level low anti-hack disable
    security-mgmt 1 state enable mode forward protocol web https
    security-mgmt 998 state enable mode forward ingress-type lan protocol web https
    security-mgmt 999 state enable ingress-type lan protocol ftp telnet ssh snmp tr069
    exit
!
end
